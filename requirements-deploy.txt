# ============================================================
# Krishi AI — Deployment requirements (CPU / cloud platforms)
# ============================================================
# torch and torchvision are NOT installed on deployment.
# All inference is handled by ONNX Runtime — the inference engine
# auto-detects torch absence and runs the pure-numpy ONNX path.
# Benefits:
#   - Image size ~200 MB instead of 1 GB+
#   - No torch/numpy ABI conflicts (RecursionError in _dtype.py)
#   - Faster cold starts on Render / Railway free tier
# ============================================================

# API server
fastapi>=0.111.0
uvicorn[standard]>=0.29.0
python-multipart>=0.0.9

# Image processing
Pillow>=10.3.0
# Exact pin: 1.26.4 is the last stable numpy 1.x release.
# onnxruntime 1.20+ can pull numpy 2.x which triggers a dtype
# recursion crash (RecursionError in _dtype._scalar_str) on Python 3.12.
numpy==1.26.4

# Configuration & serialisation
pydantic>=2.7.0
pydantic-settings>=2.3.0
python-dotenv>=1.0.1

# ONNX inference — sole inference backend on CPU deployments
onnx>=1.16.0
# Pin to <1.21.0: versions 1.21+ require numpy 2.x and conflict with pin above.
onnxruntime>=1.18.0,<1.21.0

# Async file responses
aiofiles>=23.2.0
